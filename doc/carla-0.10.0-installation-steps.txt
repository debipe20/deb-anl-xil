# Chatgpt Installation steps: 
1. Install System Dependencies
- Run the following commands to install required dependencies:

sudo apt install -y build-essential clang-12 lld-12 cmake ninja-build 
sudo apt install -y python3-dev python3-pip python3-venv libpng-dev libtiff-dev libjpeg-dev 
sudo apt update && sudo apt upgrade -y
sudo apt install libunwind-dev -y
sudo apt install -y libfreetype6-dev libgstreamer1.0-dev libgstreamer-plugins-base1.0-dev 
sudo apt install -y freeglut3-dev libopenscenegraph-dev libjsoncpp-dev
sudo apt install -y libboost-all-dev libeigen3-dev libxerces-c-dev libproj-dev

- Verify Python and CMake versions:
python3 --version  # Should be 3.8 or later
cmake --version    # Should be 3.16 or later

2. Install Unreal Engine 5.5
2.1. Clone Unreal Engine Source Code
-Ensure you have GitHub access to Epic Games’ Unreal Engine repository.

git clone --depth=1 -b 5.5 https://github.com/EpicGames/UnrealEngine.git
cd UnrealEngine

2.2. Install Unreal Engine Dependencies
./Setup.sh
./GenerateProjectFiles.sh
make

3. Clone and Build CARLA 0.10.0
3.1. Clone the CARLA Repository

cd ~
git clone --recurse-submodules --branch 0.10.0 https://github.com/carla-simulator/carla.git
git clone -b ue5-dev https://github.com/carla-simulator/carla.git CarlaUE5

git clone https://github.com/carla-simulator/carla
cd carla


3.2. Set Up CARLA Dependencies
make setup

3.3. Build CARLA Python API
make PythonAPI

3.4. Compile CARLA
make launch

4. Run CARLA Simulator
./CarlaUE5.sh -RenderOffScreen
./CarlaUE5.sh
